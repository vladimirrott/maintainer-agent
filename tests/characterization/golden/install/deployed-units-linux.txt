=== ./maintainer-alert@.service ===
[Unit]
Description=maintainer: alert for a failed %i

[Service]
Type=oneshot
Environment=DISPLAY=:1
Environment=DBUS_SESSION_BUS_ADDRESS=unix:path=%t/bus
# The script, not an inline shell command. systemd refused the inline version
# with "Unbalanced quoting, ignoring", which it reports at load time and which
# `systemd-analyze verify` prints while still exiting 0.
ExecStart=%h/.local/share/maintainer/alert.sh %i
=== ./maintainer@.service ===
[Unit]
Description=maintainer: %i
# Rootless podman cannot ESTABLISH a user namespace under the hardening
# below (newuidmap is setuid; NoNewPrivileges and RestrictSUIDSGID block
# it), but it can REUSE one. Measured on this host: cold + full hardening
# failed 10/10, warm + full hardening succeeded. So an unhardened unit
# running one fixed command establishes it first, and this unit stays
# hardened. Without the ordering, container verification would silently
# become unavailable after a reboot, and a merge gate that quietly stops
# verifying still merges.
Wants=podman-userns-warmup.service
After=podman-userns-warmup.service
OnFailure=maintainer-alert@%n.service

[Service]
Type=oneshot
# The profile decides the repository; run.sh cds into it. This unit is
# instantiated as maintainer@<profile>-<task> and must stay profile-agnostic.
WorkingDirectory=%h
Environment=DISPLAY=:1
Environment=DBUS_SESSION_BUS_ADDRESS=unix:path=%t/bus
Environment=PATH=%h/.local/bin:/usr/local/bin:/usr/bin:/bin
# %i is "<profile>-<task>", split by the wrapper.
ExecStart=%h/.local/share/maintainer/run-instance.sh %i
# 75 is EX_TEMPFAIL: the run could not ask (no network, another run holding the
# lock), recorded it in failed-<task>.json, and the catch-up slot in this
# profile's timer retries it. Without this, systemd calls that a unit failure
# and OnFailure raises a second critical popup on top of the one run.sh already
# sent. 2026-09-08: six such failures, up to twelve popups, every one of them
# self-healing.
SuccessExitStatus=75
# 5400 killed a magent review at 11:58 -> 13:28 on 2026-09-08 just as it
# finished, and because it held the profile lock the whole time, the 12:20
# issues run then timed out waiting. Reviews of this repository grew; the
# ceiling had not.
#
# LOCK_WAIT (lib/defaults.env, 3600) is deliberately SHORTER than this. A run
# waiting for the shared lock spends this same budget, so a wait as long as the
# budget would be killed by systemd before it could start. The cost is that a
# run behind a three-hour review gives up after an hour, records an exit-75
# failure, and the catch-up slot retries it. tests/run-tests.sh fails if
# LOCK_WAIT ever reaches this figure.
TimeoutStartSec=10800

# --- Containment -----------------------------------------------------------
# Mount-based isolation is NOT available here, and the directives for it were
# removed rather than left in place looking protective. Verified 2026-09-02 on
# systemd 255 / Ubuntu 24.04: ProtectHome, BindPaths and InaccessiblePaths are
# silently ignored for *user* units, and bwrap fails with "setting up uid map:
# Permission denied", both because kernel.apparmor_restrict_unprivileged_userns
# is 1. A directive that no-ops is worse than none: it reads as a guarantee.
#
# What actually contains a run:
#   1. NoNewPrivileges below, which is seccomp-based and verified to take effect.
#   2. The backend's own deny list (Claude only; see lib/backends/codex.sh).
#   3. `maintainer screen`, which refuses to BUILD a PR that could execute
#      contributor code (build.rs, dependency changes, workflow changes).
#   4. podman for anything that must run untrusted code.
#
# ProtectKernelTunables, ProtectControlGroups and RestrictNamespaces are
# deliberately ABSENT: each one breaks rootless podman, and podman is the only
# working sandbox for running contributor code on this host.
NoNewPrivileges=yes
RestrictSUIDSGID=yes
LockPersonality=yes
RestrictRealtime=yes
=== ./maintainer@magent-issues.timer ===
[Unit]
Description=maintainer: magent issues (timer)

[Timer]
# Two slots, not one. A run that cannot reach GitHub exits 75 having done
# nothing, and systemd will not retry a Type=oneshot unit, so a single slot
# turns one refused start into a whole lost cycle: two days for issues, three
# for ci, five for audit. On this host the network is down overnight and
# returns between 10:00 and 12:30, and the 09:13 slot fired into that window on
# 2026-09-04, 05, 06 and 07 without once succeeding.
#
# The second slot costs nothing on a good day. min_gate in run.sh skips a task
# whose last-<task>.json is younger than MIN_HOURS_<task>, and `finish` writes
# that file only on success, so the retry runs exactly when the first slot
# failed. tests/run-tests.sh enforces that one gap here stays under MIN_HOURS.
OnCalendar=*-*-* 12:17:00
OnCalendar=*-*-* 17:17:00
Persistent=true
RandomizedDelaySec=300
Unit=maintainer@magent-issues.service

[Install]
WantedBy=timers.target

=== ./podman-userns-warmup.service ===
[Unit]
Description=maintainer: establish the rootless podman user namespace
Documentation=man:podman(1)

[Service]
Type=oneshot
RemainAfterExit=no
Environment=PATH=%h/.local/bin:/usr/local/bin:/usr/bin:/bin
# `podman info` is enough to make podman create its pause process and map the
# subuid range. Nothing untrusted runs here.
ExecStart=/usr/bin/podman info --format {{.Host.Security.Rootless}}

# DELIBERATELY UNHARDENED, and this is the whole point of the unit.
#
# Rootless podman maps its user namespace through /usr/bin/newuidmap, which is
# setuid. NoNewPrivileges and RestrictSUIDSGID both block that, so a hardened
# unit cannot ESTABLISH a namespace. Measured on this host: cold + full
# hardening fails 10/10 with "newuidmap: write to uid_map failed"; warm + full
# hardening succeeds, because podman reuses the pause process rather than
# remapping.
#
# So the split is: this unit, which runs one fixed command and no model output,
# establishes the namespace. The maintainer unit stays fully hardened and
# reuses it. Relying on a warm namespace without this would make container
# verification silently unavailable after a reboot, which is worse than not
# having it: a merge gate that quietly stops verifying still merges.
